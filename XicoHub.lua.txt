local function GetCurrentSea()
    -- Detect which sea the player is in based on PlaceId or game state
    local placeId = game.PlaceId
    -- Sea 2 PlaceId: 4442272183
    -- Sea 3 PlaceId: 7449423635 (common known IDs)
    if placeId == 4442272183 then
        return "Sea2"
    elseif placeId == 7449423635 or placeId == 11100731498 then
        return "Sea3"
    else
        -- Fallback: try to detect by world content or location
        local map = Workspace:FindFirstChild("Map") or Workspace:FindFirstChild("Island")
        -- Check for Sea 3 specific objects
        if Workspace:FindFirstChild("ThirdSea") or Workspace:FindFirstChild("Sea3") then
            return "Sea3"
        end
        return "Sea2" -- default fallback
    end
end

local function SafeHttpGet(url)
    local ok, result = pcall(function()
        return game:HttpGet(url)
    end)
    if ok then return result end
    return nil
end

local function ParseServers(jsonStr)
    local ok, data = pcall(function()
        return HttpService:JSONDecode(jsonStr)
    end)
    if ok and type(data) == "table" then
        return data
    end
    return {}
end

-- Sort servers by age ascending (youngest first)
local function SortByAge(servers)
    table.sort(servers, function(a, b)
        local ageA = tonumber(a.Age or a.age) or 999999
        local ageB = tonumber(b.Age or b.age) or 999999
        return ageA < ageB
    end)
    return servers
end

-- Teleport via TeleportService
local function HopToServer(placeId, jobId)
    local TeleportService = game:GetService("TeleportService")
    local ok, err = pcall(function()
        TeleportService:TeleportToPlaceInstance(placeId, jobId, LocalPlayer)
    end)
    if not ok then
        warn("[XicoHub] Teleport failed: " .. tostring(err))
        Window:Notify({
            Title       = "Hop Failed",
            Description = "Could not teleport. Try again.",
            Duration    = 3,
        })
    end
end

-- Fetch server list from NightHub API for a specific boss
local function FetchServers(bossAPI)
    local url = "http://nighthub.site/boss/" .. bossAPI
    local raw = SafeHttpGet(url)
    if not raw then return {} end

    local servers = ParseServers(raw)
    -- Filter by current sea if Sea field exists
    local filtered = {}
    for _, s in ipairs(servers) do
        local sea = s.Sea or s.sea or ""
        if sea == "" or sea == CURRENT_SEA then
            table.insert(filtered, s)
        end
    end
    SortByAge(filtered)
    return filtered
end

-- Tween to position
local function TweenToPosition(targetPos, duration)
    local hrp = GetHRP()
    if not hrp then return end
    duration = duration or 0.15

    local tween = TweenService:Create(hrp, TweenInfo.new(duration, Enum.EasingStyle.Linear), {
        CFrame = CFrame.new(targetPos)
    })
    tween:Play()
    tween.Completed:Wait()
end

-- Find boss by name in workspace
local function FindBoss(bossName)
    -- Search NPCs folder and workspace
    local folders = { "NPC", "Enemies", "Bosses", Workspace }
    local lowerName = bossName:lower():gsub("%s", ""):gsub("_", "")

    -- Try to find in common NPC locations
    local function SearchIn(parent)
        for _, obj in ipairs(parent:GetChildren()) do
            if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") then
                local name = obj.Name:lower():gsub("%s", ""):gsub("_", "")
                if name:find(lowerName) or lowerName:find(name) then
                    return obj
                end
            end
        end
        return nil
    end

    for _, fname in ipairs({"NPC", "Enemies", "Bosses", "Mobs"}) do
        local folder = Workspace:FindFirstChild(fname, true)
        if folder then
            local found = SearchIn(folder)
            if found then return found end
        end
    end

    -- Fallback: search entire workspace
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") then
            local name = obj.Name:lower():gsub("%s", ""):gsub("_", "")
            if name:find(lowerName) or lowerName:find(name) then
                local hum = obj:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    return obj
                end
            end
        end
    end
    return nil
end

-- Get target part CFrame from boss
local function GetTargetPartCFrame(boss, partName)
    local part = boss:FindFirstChild(partName) or boss:FindFirstChild("HumanoidRootPart") or boss.PrimaryPart
    if part then return part.CFrame end
    return boss:GetModelCFrame()
end

-- Use selected attack tool
local function AttackWithTool(toolType)
    local char = GetCharacter()
    if not char then return end

    if toolType == "Melee" then
        -- Simulate punch/melee via remote or tool activation
        local tool = LocalPlayer.Backpack:FindFirstChild("Godhuman") or
                     LocalPlayer.Backpack:FindFirstChild("Superhuman") or
                     LocalPlayer.Backpack:FindFirstChild("Dragon Talon") or
                     LocalPlayer.Backpack:FindFirstChildWhichIsA("Tool")
        if not tool then
            -- Use bare hands attack if no melee tool
            tool = char:FindFirstChildWhichIsA("Tool")
        end
        if tool then
            if tool.Parent ~= char then
                tool.Parent = char
            end
            -- Fire tool activation
            local activate = tool:FindFirstChild("Activate") or tool:FindFirstChildOfClass("RemoteEvent")
            if activate then
                pcall(function() activate:FireServer() end)
            else
                -- Use LocalScript activation method
                pcall(function()
                    tool:Activate()
                end)
            end
        end

    elseif toolType == "Sword" then
        -- Find and use sword tool
        for _, tool in ipairs(LocalPlayer.Backpack:GetChildren()) do
            if tool:IsA("Tool") and (tool:FindFirstChild("SwordValue") or
               tool.Name:lower():find("sword") or tool.Name:lower():find("blade") or
               tool.Name:lower():find("katana") or tool.Name:lower():find("saber")) then
                if tool.Parent ~= char then tool.Parent = char end
                pcall(function() tool:Activate() end)
                return
            end
        end
        -- Fallback to any tool
        local anyTool = char:FindFirstChildWhichIsA("Tool")
        if anyTool then pcall(function() anyTool:Activate() end) end

    elseif toolType == "Fruit" then
        -- Use devil fruit moves
        for _, tool in ipairs(char:GetChildren()) do
            if tool:IsA("Tool") and (tool:FindFirstChild("FruitValue") or
               tool.Name:lower():find("fruit") or tool.Name:lower():find("phoenix") or
               tool.Name:lower():find("dough") or tool.Name:lower():find("buddha")) then
                pcall(function() tool:Activate() end)
                return
            end
        end
    end
end

-- Fast attack loop with tween movement
local function FarmBossLoop(bossName, toolType, targetPart)
    if Config.FarmingActive then return end
    Config.FarmingActive = true

    Window:Notify({
        Title       = "Farming Started",
        Description = "Farming: " .. bossName,
        Duration    = 3,
        ColoredWords = {
            { Text = bossName, Color = Color3.fromRGB(255, 200, 50) },
        },
    })

    local thread = task.spawn(function()
        while Config.FarmingActive do
            local boss = FindBoss(bossName)
            if boss then
                local bossHum = boss:FindFirstChildOfClass("Humanoid")
                if bossHum and bossHum.Health > 0 then
                    -- Tween close to boss
                    local targetCFrame = GetTargetPartCFrame(boss, targetPart)
                    local targetPos = targetCFrame.Position + (targetCFrame.LookVector * -5)

                    -- Fast tween approach
                    local hrp = GetHRP()
                    if hrp then
                        local dist = (hrp.Position - targetCFrame.Position).Magnitude
                        if dist > 8 then
                            TweenToPosition(targetPos, math.clamp(dist / 150, 0.05, 0.3))
                        end
                    end

                    -- Fast attack burst
                    for i = 1, 5 do
                        if not Config.FarmingActive then break end
                        AttackWithTool(toolType)
                        task.wait(0.05) -- ultra-fast attack
                    end

                    -- Check if boss died (stop-on-legendary handling)
                    if Config.StopOnLegendary then
                        -- Check inventory for legendary drops (basic check)
                        local hasLegendary = false
                        for _, item in ipairs(LocalPlayer.Backpack:GetChildren()) do
                            if item.Name:lower():find("legendary") or item.Name:lower():find("dragon") then
                                hasLegendary = true
                                break
                            end
                        end
                        if hasLegendary then
                            Config.FarmingActive = false
                            Window:Notify({
                                Title       = "Legendary Obtained!",
                                Description = "Stopped farming - legendary item detected.",
                                Duration    = 5,
                            })
                            break
                        end
                    end
                else
                    -- Boss not alive, wait for respawn
                    task.wait(2)
                end
            else
                -- Boss not found in server
                task.wait(3)
            end
            task.wait(0.02)
        end
    end)

    table.insert(HopThreads, thread)
end

local function StopFarming()
    Config.FarmingActive = false
    for _, t in ipairs(HopThreads) do
        pcall(task.cancel, t)
    end
    HopThreads = {}
    Window:Notify({
        Title       = "Farming Stopped",
        Description = "Auto kill has been disabled.",
        Duration    = 3,
    })
end

        Description = "Click to load " .. boss.Name .. " servers (sorted newest first)",
        Callback    = function()
            Window:Notify({
                Title       = "Refreshing...",
                Description = "Loading " .. boss.Name .. " servers",
                Duration    = 2,
            })

            task.spawn(function()
                local servers = FetchServers(boss.API)
                HopData[boss.API] = servers

                if #servers == 0 then
                    Window:Notify({
                        Title       = "No Servers",
                        Description = "No active " .. boss.Name .. " servers found.",
                        Duration    = 4,
                    })
                    return
                end

                -- Animated notification with count
                Window:Notify({
                    Title       = "✅ " .. boss.Name,
                    Description = "Found " .. #servers .. " server(s)! Best: Age " ..
                                  tostring(servers[1] and (servers[1].Age or servers[1].age) or "?") ..
                                  " | Players: " .. tostring(servers[1] and (servers[1].Players or servers[1].players) or "?"),
                    Duration    = 6,
                    ColoredWords = {
                        { Text = tostring(#servers), Color = Color3.fromRGB(0, 255, 128) },
                        { Text = "server(s)!", Color = Color3.fromRGB(255, 255, 100) },
                    },
                })

                -- Show best servers info
                for i = 1, math.min(3, #servers) do
                    local s = servers[i]
                    task.wait(0.5)
                    Window:Notify({
                        Title       = "Server #" .. i .. " - " .. boss.Name,
                        Description = "Age: " .. tostring(s.Age or s.age or "?") ..
                                      " | Players: " .. tostring(s.Players or s.players or "?") ..
                                      "\nClick 'Join #" .. i .. "' below to hop",
                        Duration    = 5,
                    })
                end
            end)
        end,
    })

    -- Direct hop buttons (top 3 servers)
    for i = 1, 3 do
        tab:CreateButton({
            Title       = "  ↳ Join Best Server #" .. i .. " - " .. boss.Name,
            Description = "Hop to server " .. i .. " (Age/Players shown after refresh)",
            Callback    = function()
                local servers = HopData[boss.API]
                if not servers or #servers == 0 then
                    -- Auto-fetch if not loaded
                    Window:Notify({
                        Title       = "Fetching...",
                        Description = "Loading servers first...",
                        Duration    = 2,
                    })
                    task.spawn(function()
                        servers = FetchServers(boss.API)
                        HopData[boss.API] = servers
                        if servers and servers[i] then
                            local s = servers[i]
                            local placeId = tonumber(s.PlaceId or s.placeId) or game.PlaceId
                            local jobId   = s.JobId or s.jobId or s.job_id or ""
                            Window:Notify({
                                Title       = "Hopping!",
                                Description = "Joining " .. boss.Name .. " server #" .. i,
                                Duration    = 3,
                            })
                            task.wait(1)
                            if jobId ~= "" then HopToServer(placeId, jobId) end
                        else
                            Window:Notify({
                                Title       = "Not Found",
                                Description = "Server #" .. i .. " not available.",
                                Duration    = 3,
                            })
                        end
                    end)
                    return
                end

                if servers[i] then
                    local s = servers[i]
                    local placeId = tonumber(s.PlaceId or s.placeId) or game.PlaceId
                    local jobId   = s.JobId or s.jobId or s.job_id or ""
                    Window:Notify({
                        Title       = "Hopping!",
                        Description = "Joining " .. boss.Name .. " server (Age: " .. tostring(s.Age or s.age or "?") .. ")",
                        Duration    = 3,
                    })
                    task.wait(0.5)
                    if jobId ~= "" then HopToServer(placeId, jobId) end
                else
                    Window:Notify({
                        Title       = "Not Enough Servers",
                        Description = "Only " .. (HopData[boss.API] and #HopData[boss.API] or 0) .. " server(s) found.",
                        Duration    = 3,
                    })
                end
            end,
        })
    end
end
